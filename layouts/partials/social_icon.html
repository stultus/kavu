{{- /* Inline SVG lookup for footer social links. Input: a Site.Params.social entry (dict with .name, .link, .icon). The lookup keys off the icon string suffix (e.g. "fab fa-github" -> "github"). Falls back to a generic "external link" glyph for unrecognised icons. */ -}} {{- $icon := .icon | default "" -}} {{- $key := $icon -}} {{- if hasPrefix $icon "fab fa-" -}}{{- $key = strings.TrimPrefix "fab fa-" $icon -}}{{- end -}} {{- if hasPrefix $key "fa-" -}}{{- $key = strings.TrimPrefix "fa-" $key -}}{{- end -}} {{- $key = lower $key -}} {{- if eq $key "github" -}} {{- else if or (eq $key "twitter") (eq $key "x-twitter") (eq $key "x") -}} {{- else if or (eq $key "linkedin") (eq $key "linkedin-in") -}} {{- else if eq $key "instagram" -}} {{- else if or (eq $key "medium") (eq $key "medium-m") -}} {{- else if eq $key "facebook" -}} {{- else if eq $key "mastodon" -}} {{- else if eq $key "rss" -}} {{- else -}} {{- end -}}
